def audit_evaluation_reads(root: Path | None = None) -> list[ReadViolation]:
    """① 评估链路模块里的读调用点(应为空)。**边界见模块 docstring 的表**。"""
    root = root or package_root()
    violations: list[ReadViolation] = []
    for path, rel in _iter_module_files(root):
        if not _matches_prefix(rel, EVALUATION_MODULES):
            continue
        for lineno, call in _read_calls_in(path):
            violations.append(ReadViolation(module=rel, lineno=lineno, call=call))
    return violations


def audit_undeclared_reads(root: Path | None = None) -> list[ReadViolation]:
    """② **全包**里含读调用、却没在 `DECLARED_READ_SITES` 逐文件声明理由的模块(应为空)。

    这道门是 ① 的补位:① 只管评估链路,而评审实测的绕过恰恰是「把读放进一个已分类的
    非评估包」。逐文件声明让**新增读文件**必须有人写一行理由。
    """
    root = root or package_root()
    violations: list[ReadViolation] = []
    for path, rel in _iter_module_files(root):
        if rel in DECLARED_READ_SITES:      # ← 门② 有逐文件豁免;门① 没有任何豁免入口
            continue
